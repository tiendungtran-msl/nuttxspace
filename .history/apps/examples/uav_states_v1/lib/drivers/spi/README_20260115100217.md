🚀 Quick Start
1. Include Header
C++
#include "lib/drivers/spi/spi_device.hpp"

using namespace drivers::spi;
2. Create Your Driver
C++
class MySensor : public Device {
public: 
    MySensor(uint8_t bus, uint32_t cs) 
        : Device(
            Config{bus, cs, SPIDEV_MODE3, 1000000},  // 1 MHz
            0x01)  // Device type
    {}
    
    int initialize() {
        if (init() != 0) return -1;
        
        // Configure sensor
        write_reg(0x6B, 0x00);  // Wake up
        
        // Switch to fast mode
        set_frequency(10000000);  // 10 MHz
        
        return 0;
    }
    
    int read_data(uint8_t *buffer) {
        return read_burst(0x3B, buffer, 14);
    }
    
protected: 
    int probe() override {
        uint8_t id;
        if (read_reg(0x75, id) != 0) return -EIO;
        if (id != 0x68) return -ENODEV;
        return 0;
    }
};
3. Use in Application
C++
MySensor sensor(1, SPIDEV_IMU);

sensor.initialize();

uint8_t data[14];
sensor.read_data(data);
📖 API Reference
Configuration
C++
struct Config {
    uint8_t bus;            // SPI bus (1, 2, 3...)
    uint32_t devid;         // Chip select ID
    spi_mode_e mode;        // SPIDEV_MODE0/1/2/3
    uint32_t frequency;     // Clock Hz
    LockMode lock_mode;     // NONE/THREADS/PREEMPTION
};

enum class LockMode {
    NONE,           // No locking (ISR-safe)
    THREADS,        // Mutex (default)
    PREEMPTION      // Critical section
};
Core Methods
Method	Description	Example

Register Access
Method	Description	Example
read_reg(reg, value)	Read 1 byte	read_reg(0x75, whoami);
read_reg(reg, buf, len)	Read N bytes	read_reg(0x3B, buf, 6);
write_reg(reg, val)	Write 1 byte	write_reg(0x6B, 0x00);
write_reg_verified(...)	Write + verify	write_reg_verified(0x6B, 0x00);
modify_reg(reg, clr, set)	Modify bits	modify_reg(0x6A, 0x08, 0x40);
read_burst(reg, buf, len)	Fast multi-read	read_burst(0x3B, buf, 14);

💡 Common Patterns
Pattern 1: Simple Read/Write
C++
// Read WHO_AM_I
uint8_t id;
read_reg(0x75, id);

// Write power management
write_reg(0x6B, 0x00);
Pattern 2: Burst Read (Fast)
C++
uint8_t buffer[14];

// Read 14 bytes from 0x3B
read_burst(0x3B, buffer, 14);

// Parse
int16_t accel_x = (buffer[0] << 8) | buffer[1];
Pattern 3: Modify Bits
C++
// Set bit 6, clear bit 3 in register 0x6A
modify_reg(0x6A, 
           (1 << 3),  // Clear mask
           (1 << 6)); // Set mask
Pattern 4: Write with Verification
C++
// Write and auto-verify
if (write_reg_verified(0x1C, 0x10) != 0) {
    printf("Verify failed\n");
}
Pattern 5: Speed Switching
C++
// Init at 1 MHz
set_frequency(1000000);
write_reg(0x6B, 0x00);
write_reg(0x1C, 0x10);

// Read at 10 MHz
set_frequency(10000000);
read_burst(0x3B, buffer, 14);
Pattern 6: ISR-Safe Read
C++
class FastIMU :  public Device {
public:
    FastIMU() : Device(
        Config{1, SPIDEV_IMU, SPIDEV_MODE3, 20000000, 
               LockMode::NONE},  // ← No locking
        0x01) {}
    
    void timer_isr() {
        // Safe to call from ISR
        read_burst(0x3B, _buffer, 14);
    }
};

        uint8_t val;

✅ Best Practices
DO
C++
✅ Check return values
if (init() != 0) return -1;

✅ Use burst reads
read_burst(0x3B, buf, 14);  // Fast

✅ Verify critical writes
write_reg_verified(0x6B, 0x00);

✅ Switch frequency
set_frequency(1000000);   // Init
set_frequency(10000000);  // Read
DON'T
C++
❌ Ignore errors
init();  // What if it fails? 

❌ Multiple single reads
read_reg(0x3B, &b0, 1);
read_reg(0x3C, &b1, 1);  // Use read_burst()! 

❌ Call from ISR with THREADS lock
LockMode::THREADS + ISR = DEADLOCK

❌ Use exceptions
throw std::exception();  // Disabled in embedded
📋 Complete Example
C++
// drivers/imu/mpu6050.hpp
#pragma once
#include "../../lib/drivers/spi/spi_device.hpp"

namespace drivers {
namespace imu {

class MPU6050 : public spi::Device {
public: 
    struct Data {
        float accel[3];
        float gyro[3];
        uint64_t timestamp_us;
    };
    
    MPU6050(uint8_t bus, uint32_t cs)
        : Device(Config{bus, cs, SPIDEV_MODE3, 1000000}, 0x09)
        , _accel_scale(9.81f / 4096.0f)
        , _gyro_scale(0.0174533f / 16.4f)
    {}
    
    int initialize() {
        if (init() != 0) return -1;
        
        // Reset
        write_reg(0x6B, 0x80);
        usleep(100000);
        
        // Wake up
        write_reg_verified(0x6B, 0x00);
        
        // Configure
        write_reg(0x1B, 0x18);  // ±2000 dps
        write_reg(0x1C, 0x10);  // ±8g
        
        // Fast mode
        set_frequency(10000000);
        
        return 0;
    }
    
    int read(Data &data) {
        uint8_t buf[14];
        
        if (read_burst(0x3B, buf, 14) != 0) {
            return -EIO;
        }
        
        data.timestamp_us = hrt_absolute_time();
        
        for (int i = 0; i < 3; i++) {
            int16_t raw_a = (buf[i*2] << 8) | buf[i*2+1];
            int16_t raw_g = (buf[8+i*2] << 8) | buf[8+i*2+1];
            
            data.accel[i] = raw_a * _accel_scale;
            data.gyro[i] = raw_g * _gyro_scale;
        }
        
        return 0;
    }
    
protected:
    int probe() override {
        uint8_t id;
        if (read_reg(0x75, id) != 0) return -EIO;
        if (id != 0x68) return -ENODEV;
        return 0;
    }
    
private:
    float _accel_scale;
    float _gyro_scale;
};

} // namespace imu
} // namespace drivers
C++
// Usage
#include "drivers/imu/mpu6050.hpp"

int main() {
    drivers::imu::MPU6050 imu(1, SPIDEV_IMU);
    
    if (imu.initialize() != 0) {
        printf("IMU init failed\n");
        return -1;
    }
    
    while (true) {
        drivers::imu::MPU6050::Data data;
        
        if (imu.read(data) == 0) {
            printf("Accel: %. 2f, %. 2f, %.2f m/s²\n",
                   data.accel[0], data.accel[1], data.accel[2]);
        }
        
        usleep(10000);
    }
}
